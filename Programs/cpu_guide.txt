 * Stack Pointer: 0x0007FFF0

 * SYSTEM MEMORY MAP:
 * Region           | Base Address | End Address | Size   | Access | Description
 * -----------------|--------------|-------------|--------|--------|------------------------------
 * Boot ROM         | 0x00000000   | 0x0003FFFF  | 256 KB | R      | Wozmon
 * Unified RAM      | 0x00040000   | 0x0007FFFF  | 256 KB | R/W    | Program Executable & Stack
 * Standard MMIO    | 0x00080000   | 0x00081FFF  | 8 KB   | R/W    | Hardware Registers
 * Graphics Page 0  | 0x00082000   | 0x000A13FF  | 128 KB | R/W    | Front Framebuffer (320x200, 16-bit RGB565)
 * Graphics Page 1  | 0x000A1400   | 0x000C07FF  | 128 KB | R/W    | Back Framebuffer (320x200, 16-bit RGB565)
 *
 * MMIO REGISTERS:
 * PAGE_REG (0x00080000): Write 0 for Page 0, 1 for Page 1 display presentation.
 * TERM_OUT (0x00080004): Write ASCII character byte to output to terminal.
 * KEY_DATA (0x00080020): Read ASCII byte from keyboard buffer FIFO.
 * KEY_STAT (0x00080024): Read status (1 = Key Available, 0 = Buffer Empty).
 * GRAPHIC_RAM (0x00082000): Start of 16-bit packed VRAM array.
 *
 * COLOR FORMAT (RGB565):
 * GRAPHIC_RAM uses standard 16-bit RGB565 color format (5 bits Red, 6 bits Green, 5 bits Blue).
 * - Red:   Bits [15:11] (0 to 31)
 * - Green: Bits [10:5]  (0 to 63)
 * - Blue:  Bits [4:0]   (0 to 31)

 * This guide serves as a template reference. Because programs compile with

 * * The math functions included below (__mulsi3, __divsi3, __modsi3, __muldi3, __divdi3)

#include <stdint.h>

#define PAGE_REG     ((volatile uint32_t *)0x00080000)
#define TERM_OUT     ((volatile uint32_t *)0x00080004)
#define KEY_DATA     ((volatile uint32_t *)0x00080020)
#define KEY_STAT     ((volatile uint32_t *)0x00080024)
#define GRAPHIC_RAM  ((volatile uint16_t *)0x00082000)

#define SCREEN_WIDTH  320
#define SCREEN_HEIGHT 200
#define PAGE_PIXELS   (SCREEN_WIDTH * SCREEN_HEIGHT) // 64,000 pixels per page

// RGB565 Macro: r (0..31), g (0..63), b (0..31)
#define MAKE_RGB(r, g, b) ((uint16_t)((((r) & 0x1F) << 11) | (((g) & 0x3F) << 5) | ((b) & 0x1F)))

#define COLOR_BLACK  ((uint16_t)0x0000)
#define COLOR_WHITE  ((uint16_t)0xFFFF)
#define COLOR_RED    ((uint16_t)0xF800)
#define COLOR_GREEN  ((uint16_t)0x07E0)
#define COLOR_BLUE   ((uint16_t)0x001F)

// ----------------------------------------------------------------------------
// MATH HELPER TEMPLATES
// ----------------------------------------------------------------------------

int __mulsi3(int a, int b) {
	int res = 0;
	int neg = (a < 0) ^ (b < 0);
	unsigned int ua = (a < 0) ? -a : a;
	unsigned int ub = (b < 0) ? -b : b;
	while (ub > 0) {
		if (ub & 1) res += ua;
		ua <<= 1;
		ub >>= 1;
	}
	return neg ? -res : res;
}

int __divsi3(int a, int b) {
	if (b == 0) return 0;
	int neg = (a < 0) ^ (b < 0);
	unsigned int num = (a < 0) ? -a : a;
	unsigned int den = (b < 0) ? -b : b;
	unsigned int quot = 0, rem = 0;
	for (int i = 31; i >= 0; i--) {
		rem = (rem << 1) | ((num >> i) & 1);
		if (rem >= den) {
			rem -= den;
			quot |= (1U << i);
		}
	}
	return neg ? -(int)quot : (int)quot;
}

int __modsi3(int a, int b) {
	if (b == 0) return 0;
	int q = __divsi3(a, b);
	return a - __mulsi3(q, b);
}

int64_t __muldi3(int64_t a, int64_t b) {
	int neg = (a < 0) ^ (b < 0);
	uint64_t ua = (a < 0) ? -a : a;
	uint64_t ub = (b < 0) ? -b : b;
	uint64_t res = 0;
	while (ub > 0) {
		if (ub & 1) res += ua;
		ua <<= 1;
		ub >>= 1;
	}
	return neg ? -(int64_t)res : (int64_t)res;
}

int64_t __divdi3(int64_t a, int64_t b) {
	if (b == 0) return 0;
	int neg = (a < 0) ^ (b < 0);
	uint64_t num = (a < 0) ? -a : a;
	uint64_t den = (b < 0) ? -b : b;
	uint64_t quot = 0, rem = 0;
	for (int i = 63; i >= 0; i--) {
		rem = (rem << 1) | ((num >> i) & 1);
		if (rem >= den) {
			rem -= den;
			quot |= (1ULL << i);
		}
	}
	return neg ? -(int64_t)quot : (int64_t)quot;
}

// ----------------------------------------------------------------------------
// FRAMEBUFFER GRAPHICS DRAWING HELPERS

static inline void put_pixel(int page, int x, int y, uint16_t color) {
	if (x >= 0 && x < SCREEN_WIDTH && y >= 0 && y < SCREEN_HEIGHT) {
		uint32_t offset = (page * PAGE_PIXELS) + (y * SCREEN_WIDTH) + x;
		GRAPHIC_RAM[offset] = color;
	}
}

void clear_screen(int page, uint16_t color) {
	uint32_t offset = page * PAGE_PIXELS;
	for (int i = 0; i < PAGE_PIXELS; i++) {
		GRAPHIC_RAM[offset + i] = color;
	}
}

void print_string(const char *str) {
	while (*str) {
		*TERM_OUT = (uint32_t)(*str++);
	}
}

void return_to_wozmon_on_key(void) {
	print_string("\nPress any key to return to Wozmon...");
	while ((*KEY_STAT & 1) == 0);
	(void)*KEY_DATA;
	
	*PAGE_REG = 0;
	void (*wozmon_entry)(void) = (void (*)(void))0x00000000;
	wozmon_entry();
}

int main(void) {
	uint32_t active_page = 0;
	print_string("[RV32I Core] Program execution started in RAM.\n");

	clear_screen(0, COLOR_BLACK);
	clear_screen(1, COLOR_BLACK);

	while (1) {
		uint32_t back_page = active_page ^ 1;

		// Render application graphics into back_page here...

		*PAGE_REG = back_page;
		active_page = back_page;

		if (*KEY_STAT & 1) {
			break;
		}
	}

	return_to_wozmon_on_key();
	return 0;
}
